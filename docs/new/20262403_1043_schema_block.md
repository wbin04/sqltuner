# Task: 3 fixes + tích hợp Generate Schema vào Chat UI

## Context codebase

**Stack:** FastAPI + React 18 + TypeScript + Ollama (qwen2.5:3b)

**Chat flow hiện tại:**
- `chat.py` endpoint: nhận message → detect SQL bằng `_is_sql_query()` → chọn prompt

- `GenerateSchemaModal.tsx`: 3-step modal (input → clarify → preview → apply)
- Chỉ accessible từ `TableEditor.tsx` (nút "Generate with AI")

## Fix 2: JSON parse error — LLM trả về truncated/malformed JSON

**File:** `backend/app/services/llm_service.py`

**Vấn đề:** Log lỗi:
~~~
[SCHEMA-GEN] JSON parse error: Expecting ',' delimiter: line 299 column 35 (char 11476)
~~~
LLM sinh JSON dài bị cắt ở `num_predict=2048`, hoặc có trailing text sau JSON.

**Thay đổi trong method `generate_schema_from_prompt`:**

~~~python
async def generate_schema_from_prompt(
    self,
    user_description: str,
    clarifications: list[dict] = None,
) -> dict:
    prompt = get_schema_generation_prompt(user_description, clarifications)

    payload_config = OllamaPayload(
        model=self.chat_model,
        prompt=prompt,
        system_prompt=SCHEMA_GENERATION_SYSTEM_PROMPT,
        temperature=0.1,        # giảm từ 0.2 → ít hallucinate hơn
        num_ctx=3072,
        num_predict=2500,       # tăng để tránh truncate
        json_mode=True,
    )

    url = self._build_api_url()
    payload = payload_config.to_dict()
    payload["keep_alive"] = "120m"

    logger.info("[SCHEMA-GEN] Generating schema for: %s...", user_description[:80])

    raw = ""
    try:
        raw = await self._make_ollama_request(payload, url)

        # Guard 1: strip markdown fences nếu model wrap JSON
        clean = re.sub(r"```(?:json)?\s*", "", raw)
        clean = re.sub(r"```", "", clean).strip()

        # Guard 2: extract JSON object — tìm { ... } ngoài cùng
        # Dùng để loại bỏ text trước/sau JSON kể cả khi JSON bị preceded bởi explanation
        brace_start = clean.find('{')
        brace_end = clean.rfind('}')
        if brace_start != -1 and brace_end != -1 and brace_end > brace_start:
            clean = clean[brace_start:brace_end + 1]

        # Guard 3: fix truncated JSON — nếu vẫn invalid thì thử repair
        try:
            schema_json = json.loads(clean)
        except json.JSONDecodeError:
            # Truncated JSON: tìm last complete table entry
            # Cắt tại vị trí hợp lệ cuối cùng trong "tables" array
            logger.warning("[SCHEMA-GEN] JSON malformed, attempting repair...")
            schema_json = self._repair_truncated_schema_json(clean)

        if "tables" not in schema_json or not schema_json["tables"]:
            raise ValueError("LLM response missing 'tables' key or empty tables")

        logger.info(
            "[SCHEMA-GEN] Generated %s tables: %s",
            len(schema_json["tables"]),
            ", ".join(t.get("name", "") for t in schema_json["tables"]),
        )
        return schema_json

    except ValueError:
        raise
    except Exception as e:
        logger.error(f"[SCHEMA-GEN] Generation failed: {e}\nRaw snippet: {raw[:300]}")
        raise ValueError(
            "Schema generation failed: LLM returned invalid JSON. "
            "Try describing your system with fewer tables or more specific details."
        )


def _repair_truncated_schema_json(self, broken_json: str) -> dict:
    """
    Cố gắng recover JSON bị truncate giữa chừng.
    Chiến lược: tìm last complete object trong 'tables' array.
    """
    # Tìm vị trí bắt đầu tables array
    tables_start = broken_json.find('"tables"')
    if tables_start == -1:
        raise ValueError("Cannot find 'tables' key in response")

    # Tìm [ sau "tables":
    arr_start = broken_json.find('[', tables_start)
    if arr_start == -1:
        raise ValueError("Cannot find tables array start")

    # Tìm last complete } trước khi JSON bị cắt
    # Đếm bracket depth để tìm table object hoàn chỉnh cuối cùng
    depth = 0
    last_complete_end = arr_start
    i = arr_start

    while i < len(broken_json):
        ch = broken_json[i]
        if ch == '{':
            depth += 1
        elif ch == '}':
            depth -= 1
            if depth == 0:
                last_complete_end = i
        i += 1

    if last_complete_end == arr_start:
        raise ValueError("No complete table object found")

    # Rebuild JSON với tables array được đóng đúng
    partial = broken_json[:last_complete_end + 1]

    # Tìm system_name nếu có
    try:
        sys_name_match = re.search(r'"system_name"\s*:\s*"([^"]+)"', broken_json)
        system_name = sys_name_match.group(1) if sys_name_match else "Generated Schema"
    except Exception:
        system_name = "Generated Schema"

    # Tìm các table objects đã hoàn chỉnh
    tables_section = broken_json[arr_start:last_complete_end + 1]

    # Parse từng table object hoàn chỉnh
    complete_tables = []
    depth = 0
    obj_start = -1

    for i, ch in enumerate(tables_section):
        if ch == '{':
            if depth == 0:
                obj_start = i
            depth += 1
        elif ch == '}':
            depth -= 1
            if depth == 0 and obj_start != -1:
                try:
                    obj = json.loads(tables_section[obj_start:i + 1])
                    if "name" in obj and "columns" in obj:
                        complete_tables.append(obj)
                except Exception:
                    pass
                obj_start = -1

    if not complete_tables:
        raise ValueError("No parseable table objects found")

    logger.info(f"[SCHEMA-GEN] Repaired JSON: recovered {len(complete_tables)} tables")

    return {
        "system_name": system_name,
        "tables": complete_tables,
        "relationships": [],
        "design_notes": ["Schema was partially recovered due to response truncation."],
    }
~~~

### Fix 3a: Intent detection trong `backend/app/api/v1/endpoints/chat.py`

**Vấn đề:** `_is_sql_query()` chỉ detect SQL keywords, không detect intent "thiết kế schema". Khi user hỏi *"tôi muốn tạo database bán thức ăn nhanh"* → bị route vào SQL prompt → LLM sinh markdown có ````sql` block → `sql_generated` bị set → hiện Execute/Explain/Optimize buttons không phù hợp.

**Thêm function mới vào `chat.py`:**

~~~python
def _is_schema_design_intent(message: str) -> bool:
    """
    Detect xem user đang muốn THIẾT KẾ schema (tạo database mới)
    thay vì query database hiện tại.
    """
    message_lower = message.lower()

    # Keywords tiếng Việt và tiếng Anh chỉ thiết kế DB mới
    design_keywords = [
        # Tiếng Việt
        "tạo database", "tạo cơ sở dữ liệu", "thiết kế database",
        "thiết kế cơ sở dữ liệu", "tạo bảng cho", "tạo schema",
        "xây dựng database", "xây dựng cơ sở dữ liệu",
        "thiết kế bảng", "cần database", "cần cơ sở dữ liệu",
        "muốn tạo database", "muốn tạo cơ sở dữ liệu",
        "hệ thống quản lý", "muốn thiết kế",
        # Tiếng Anh
        "create a database", "design a database", "design database",
        "create database schema", "design schema for",
        "database for", "schema for", "build a database",
        "create tables for", "data model for", "erd for",
        "i need a database", "i want to create a database",
        "create a schema", "design a schema",
    ]

    for kw in design_keywords:
        if kw in message_lower:
            return True

    return False
~~~

**Tạo component mới `frontend/src/components/editor/SchemaBlock.tsx`:**

~~~tsx
/**
 * SchemaBlock — hiển thị schema được sinh từ Chat
 * Hiện trong ChatArea khi message có schema_generated
 */
import { useState } from 'react';
import { Database, ChevronDown, ChevronRight, Check, Sparkles, AlertCircle } from 'lucide-react';
import { cn } from '../../lib/utils';
import { SchemaGeneratedData } from '../../services/chatService';

interface Props {
  schema: SchemaGeneratedData;
  onApplyToSandbox: (schema: SchemaGeneratedData) => void;
  isSimulationWorkspace: boolean;
}

export function SchemaBlock({ schema, onApplyToSandbox, isSimulationWorkspace }: Props) {
  const [expandedTables, setExpandedTables] = useState<Set<string>>(new Set());
  const [confirmApply, setConfirmApply] = useState(false);

  const toggleTable = (tableName: string) => {
    setExpandedTables(prev => {
      const next = new Set(prev);
      next.has(tableName) ? next.delete(tableName) : next.add(tableName);
      return next;
    });
  };

  const handleApply = () => {
    if (!confirmApply) {
      setConfirmApply(true);
      return;
    }
    onApplyToSandbox(schema);
    setConfirmApply(false);
  };

  return (
    <div className={cn(
      'rounded-xl border overflow-hidden',
      'border-purple-200 dark:border-purple-800',
      'bg-purple-50 dark:bg-purple-950/30'
    )}>
      {/* Header */}
      <div className="flex items-center gap-2 px-4 py-3 bg-purple-100 dark:bg-purple-900/40 border-b border-purple-200 dark:border-purple-800">
        <Sparkles className="w-4 h-4 text-purple-600 dark:text-purple-400" />
        <span className="text-sm font-semibold text-purple-700 dark:text-purple-300">
          {schema.system_name}
        </span>
        <span className="ml-auto text-xs text-purple-500 dark:text-purple-400">
          {schema.tables.length} tables generated
        </span>
      </div>

      {/* Tables list */}
      <div className="divide-y divide-purple-100 dark:divide-purple-900">
        {schema.tables.map((table) => {
          const isExpanded = expandedTables.has(table.name);
          const pkCols = table.columns.filter(c => c.is_pk);
          const fkCols = table.foreign_keys || [];
          const idxs = table.indexes || [];

          return (
            <div key={table.name}>
              {/* Table header row */}
              <button
                type="button"
                onClick={() => toggleTable(table.name)}
                className="w-full flex items-center gap-3 px-4 py-2.5 hover:bg-purple-100 dark:hover:bg-purple-900/30 transition-colors text-left"
              >
                {isExpanded
                  ? <ChevronDown className="w-3.5 h-3.5 text-purple-400 flex-shrink-0" />
                  : <ChevronRight className="w-3.5 h-3.5 text-purple-400 flex-shrink-0" />
                }
                <Database className="w-3.5 h-3.5 text-blue-500 flex-shrink-0" />
                <span className="font-mono text-sm font-semibold text-blue-600 dark:text-blue-400">
                  {table.name}
                </span>
                <span className="text-xs text-gray-400 ml-1">
                  {table.columns.length} cols
                  {idxs.length > 0 && ` · ${idxs.length} idx`}
                  {fkCols.length > 0 && ` · ${fkCols.length} fk`}
                </span>
                {table.purpose && (
                  <span className="ml-auto text-xs text-gray-400 italic truncate max-w-[200px]">
                    {table.purpose}
                  </span>
                )}
              </button>

              {/* Expanded columns */}
              {isExpanded && (
                <div className="px-10 pb-3 space-y-1">
                  {table.columns.map(col => (
                    <div key={col.name} className="flex items-center gap-2 text-xs">
                      <span className={cn(
                        'font-mono',
                        col.is_pk
                          ? 'text-yellow-600 dark:text-yellow-400 font-bold'
                          : fkCols.some(fk => fk.column === col.name)
                            ? 'text-blue-600 dark:text-blue-400'
                            : 'text-gray-700 dark:text-gray-300'
                      )}>
                        {col.name}
                      </span>
                      <span className="text-gray-400 font-mono">{col.type}</span>
                      {col.is_pk && (
                        <span className="px-1 py-0.5 rounded bg-yellow-100 dark:bg-yellow-900/30 text-yellow-600 dark:text-yellow-400 text-[10px] font-bold">
                          PK
                        </span>
                      )}
                      {fkCols.some(fk => fk.column === col.name) && (
                        <span className="px-1 py-0.5 rounded bg-blue-100 dark:bg-blue-900/30 text-blue-600 dark:text-blue-400 text-[10px] font-bold">
                          FK
                        </span>
                      )}
                      {!col.is_nullable && (
                        <span className="text-gray-400 text-[10px]">NOT NULL</span>
                      )}
                    </div>
                  ))}

                  {/* FK references */}
                  {fkCols.length > 0 && (
                    <div className="mt-2 pt-2 border-t border-purple-100 dark:border-purple-900">
                      {fkCols.map((fk, i) => (
                        <div key={i} className="text-[11px] text-gray-400">
                          {fk.column} → {fk.ref_table}.{fk.ref_column}
                          {fk.on_delete && ` (ON DELETE ${fk.on_delete})`}
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              )}
            </div>
          );
        })}
      </div>

      {/* Design notes */}
      {schema.design_notes && schema.design_notes.length > 0 && (
        <div className="px-4 py-2 border-t border-purple-200 dark:border-purple-800 bg-amber-50 dark:bg-amber-900/10">
          {schema.design_notes.map((note, i) => (
            <p key={i} className="text-xs text-amber-600 dark:text-amber-400">• {note}</p>
          ))}
        </div>
      )}

      {/* Action bar */}
      <div className="flex items-center gap-2 px-4 py-3 border-t border-purple-200 dark:border-purple-800 bg-white dark:bg-gray-900">
        {isSimulationWorkspace ? (
          <>
            {confirmApply && (
              <div className="flex items-center gap-1 text-xs text-red-500 mr-2">
                <AlertCircle className="w-3.5 h-3.5" />
                Will replace current sandbox schema
              </div>
            )}
            <button
              type="button"
              onClick={handleApply}
              className={cn(
                'flex items-center gap-2 px-4 py-1.5 rounded-lg text-sm font-medium text-white transition-colors',
                confirmApply
                  ? 'bg-red-600 hover:bg-red-700'
                  : 'bg-green-600 hover:bg-green-700'
              )}
            >
              <Check className="w-3.5 h-3.5" />
              {confirmApply ? 'Confirm Apply' : 'Apply to Sandbox'}
            </button>
            {confirmApply && (
              <button
                type="button"
                onClick={() => setConfirmApply(false)}
                className="px-3 py-1.5 text-sm text-gray-500 hover:text-gray-700"
              >
                Cancel
              </button>
            )}
          </>
        ) : (
          <p className="text-xs text-gray-400 italic">
            Switch to a Simulation workspace to apply this schema
          </p>
        )}
      </div>
    </div>
  );
}
~~~

import { SchemaGeneratedData } from '../../services/chatService';
import { v4 as uuidv4 } from 'uuid';

// Thêm handler trong EditorPage
const handleApplySchemaToSandbox = async (schema: SchemaGeneratedData) => {
  if (!workspace || workspace.db_type !== DbType.SIMULATION) {
    toast.error('Please switch to a Simulation workspace to apply schema');
    return;
  }

  try {
    // Convert SchemaGeneratedData → UpdateSchemaPayload format
    const schemaPayload = {
      tables: schema.tables.map(t => ({
        name: t.name,
        columns: t.columns.map(c => ({
          name: c.name,
          type: c.type,
          is_pk: c.is_pk,
          is_nullable: c.is_nullable,
          default: c.default ?? null,
        })),
        foreign_keys: t.foreign_keys || [],
        indexes: (t.indexes || []).map(idx => ({
          name: idx.name,
          column_names: idx.column_names,
          unique: idx.unique,
        })),
        sample_data: [],
      })),
    };

    await workspaceService.updateSimulationSchema(workspaceId!, schemaPayload);
    toast.success(`Schema "${schema.system_name}" applied — ${schema.tables.length} tables created`);
  } catch (err) {
    toast.error('Failed to apply schema to sandbox');
  }
};
