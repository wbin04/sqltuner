# Task: Thêm chức năng "Generate Schema from Prompt" vào SQLTuner Sandbox

## Context về codebase hiện tại

**Stack:**
- Backend: Python 3.10 + FastAPI
- LLM: Ollama local — `coder_model` (Qwen/SQLCoder) và `chat_model` (Llama3)
- LLM client: `LLMService` trong `backend/app/services/llm_service.py`
  - Method gọi LLM: `_call_ollama(model, prompt, system_prompt, temperature, max_tokens, json_mode)`
  - JSON mode: truyền `json_mode=True` vào `OllamaPayload`
- Frontend: React 18 + TypeScript + TailwindCSS

**Luồng Sandbox hiện tại:**
1. User tạo workspace loại `SIMULATION` (db_type = "simulation")
2. Vào `TableEditor` (`/table-editor/:workspaceId`) — file `frontend/src/pages/user/TableEditor.tsx`
3. Tạo thủ công từng bảng/cột/index qua `StructureEditor`, `TablesSidebar`
4. Generate sample data qua `POST /simulation/generate-data-with-fk`
5. Save schema qua `PUT /connections/{id}/schema` → lưu vào `DBConnection.meta_schema` (JSONB)
6. Test query qua `POST /sql/execute` → chạy trên SQLite in-memory

**Schema format lưu trong `meta_schema` (JSONB):**

~~~json
{
  "tables": [
    {
      "name": "users",
      "columns": [
        { "name": "id", "type": "UUID", "is_pk": true, "is_nullable": false },
        { "name": "email", "type": "VARCHAR(255)", "is_pk": false, "is_nullable": false }
      ],
      "foreign_keys": [
        { "column": "role_id", "ref_table": "roles", "ref_column": "id" }
      ],
      "indexes": [
        { "name": "idx_users_email", "column_names": ["email"], "unique": true }
      ],
      "sample_data": []
    }
  ]
}
~~~

**Pydantic schemas đã có:**
- `SchemaDef` / `TableDef` / `ColumnDef` / `ForeignKeyDef` / `IndexDef` — trong `backend/app/schemas/schema_def.py`
- `SimulationSchema` / `SimulationTable` / `SimulationColumn` — trong `frontend/src/types/simulation.ts`

**API đã có liên quan:**
- `PUT /connections/{id}/schema` — update toàn bộ meta_schema
- `POST /simulation/generate-data-with-fk` — generate mock data theo FK
- Router: `backend/app/api/v1/api.py` — đăng ký thêm vào đây

---

## Mục tiêu

Thêm một flow mới trong `TableEditor`: user nhập mô tả hệ thống bằng ngôn ngữ tự nhiên
→ LLM sinh ra schema JSON → hiển thị preview → user confirm → apply vào sandbox
→ tự động generate sample data → sẵn sàng test query.

~~~python
SCHEMA_CLARIFICATION_SYSTEM_PROMPT = """
You are a database architect assistant.
The user wants to build a database schema but their description is too vague to generate a good schema.
Generate 2-3 focused clarifying questions to better understand their requirements.

OUTPUT FORMAT — respond ONLY with valid JSON:
{
  "needs_clarification": true,
  "questions": [
    {
      "question": "clear question text",
      "why": "one sentence explaining why this affects the schema design",
      "options": ["option A", "option B", "option C or describe your own"]
    }
  ]
}

If the description is clear enough (more than 15 words describing specific entities/actions),
respond with:
{
  "needs_clarification": false,
  "questions": []
}
"""


def get_schema_generation_prompt(
    user_description: str,
    clarifications: list[dict] = None
) -> str:
    clarification_section = ""
    if clarifications:
        pairs = "\n".join(
            f"Q: {c.get('question', '')}\nA: {c.get('answer', '')}"
            for c in clarifications
        )
        clarification_section = f"\n\nAdditional context from user:\n{pairs}"

    return f"""System description:
{user_description}{clarification_section}

Generate the database schema JSON following all DESIGN RULES.
Response (JSON only):"""


def get_schema_clarification_prompt(user_description: str) -> str:
    return f"""User wants to build: "{user_description}"

Evaluate if this description is clear enough to generate a good schema.
If not, ask 2-3 clarifying questions.
Response (JSON only):"""
~~~

~~~python
async def check_schema_clarification(
    self, user_description: str
) -> dict:
    """
    Kiểm tra xem prompt có đủ rõ để sinh schema không.
    Trả về { needs_clarification: bool, questions: list }
    """
    prompt = get_schema_clarification_prompt(user_description)

    payload_config = OllamaPayload(
        model=self.chat_model,        # dùng chat model — task này không cần SQL
        prompt=prompt,
        system_prompt=SCHEMA_CLARIFICATION_SYSTEM_PROMPT,
        temperature=0.2,
        num_ctx=1024,
        num_predict=256,
        json_mode=True,
    )

    url = self._build_api_url()
    payload = payload_config.to_dict()

    try:
        raw = await self._make_ollama_request(payload, url)
        clean = re.sub(r"```json|```", "", raw).strip()
        result = json.loads(clean)
        logger.info(
            f"[SCHEMA-GEN] Clarification check: "
            f"needs_clarification={result.get('needs_clarification')}"
        )
        return result
    except Exception as e:
        logger.warning(f"[SCHEMA-GEN] Clarification check failed: {e}")
        # Fallback: tiếp tục sinh schema luôn
        return {"needs_clarification": False, "questions": []}


async def generate_schema_from_prompt(
    self,
    user_description: str,
    clarifications: list[dict] = None,
) -> dict:
    """
    Sinh database schema từ mô tả ngôn ngữ tự nhiên.
    Trả về parsed JSON schema theo SCHEMA_GENERATION_SYSTEM_PROMPT format.
    """
    prompt = get_schema_generation_prompt(user_description, clarifications)

    payload_config = OllamaPayload(
        model=self.chat_model,        # Llama3 tốt hơn cho design task
        prompt=prompt,
        system_prompt=SCHEMA_GENERATION_SYSTEM_PROMPT,
        temperature=0.2,
        num_ctx=4096,
        num_predict=2048,             # schema JSON có thể dài
        json_mode=True,
    )

    url = self._build_api_url()
    payload = payload_config.to_dict()
    payload["keep_alive"] = "120m"

    logger.info(
        f"[SCHEMA-GEN] Generating schema for: "
        f"{user_description[:80]}..."
    )

    try:
        raw = await self._make_ollama_request(payload, url)
        clean = re.sub(r"```json|```", "", raw).strip()
        schema_json = json.loads(clean)

        # Validate tối thiểu
        if "tables" not in schema_json:
            raise ValueError("LLM response missing 'tables' key")

        logger.info(
            f"[SCHEMA-GEN] Generated {len(schema_json['tables'])} tables: "
            + ", ".join(t["name"] for t in schema_json["tables"])
        )
        return schema_json

    except json.JSONDecodeError as e:
        logger.error(f"[SCHEMA-GEN] JSON parse error: {e}\nRaw: {raw}")
        raise ValueError(f"Schema generation failed: invalid JSON from LLM")
    except Exception as e:
        logger.error(f"[SCHEMA-GEN] Generation failed: {e}")
        raise
~~~

### 3. Tạo file mới `backend/app/services/schema_generator_service.py`

~~~python
import logging
from typing import Optional
from uuid import UUID

from app.schemas.schema_def import (
    ColumnDef, ForeignKeyDef, IndexDef, SchemaDef, TableDef
)
from app.services.llm_service import llm_service

logger = logging.getLogger(__name__)


class SchemaGeneratorService:
    """
    Converts LLM-generated schema JSON → SchemaDef (Pydantic model)
    compatible với toàn bộ Sandbox pipeline hiện có.
    """

    async def check_clarification(self, description: str) -> dict:
        return await llm_service.check_schema_clarification(description)

    async def generate(
        self,
        description: str,
        clarifications: list[dict] = None,
    ) -> dict:
        """
        Sinh schema và trả về cả raw JSON lẫn SchemaDef đã validate.
        """
        raw_schema = await llm_service.generate_schema_from_prompt(
            description, clarifications
        )

        schema_def = self._convert_to_schema_def(raw_schema)

        return {
            "schema_def": schema_def,
            "raw": raw_schema,                          # dùng cho frontend preview
            "system_name": raw_schema.get("system_name", "Generated Schema"),
            "relationships": raw_schema.get("relationships", []),
            "design_notes": raw_schema.get("design_notes", []),
        }

    def _convert_to_schema_def(self, raw: dict) -> SchemaDef:
        """
        Convert LLM JSON output → SchemaDef Pydantic model.
        Ánh xạ 1-1 với format meta_schema của DBConnection.
        """
        tables = []

        for t in raw.get("tables", []):
            columns = [
                ColumnDef(
                    name=col["name"],
                    type=col["type"],
                    is_pk=col.get("is_pk", False),
                    is_nullable=col.get("is_nullable", True),
                    default=col.get("default"),
                )
                for col in t.get("columns", [])
            ]

            foreign_keys = [
                ForeignKeyDef(
                    column=fk["column"],
                    ref_table=fk["ref_table"],
                    ref_column=fk.get("ref_column", "id"),
                )
                for fk in t.get("foreign_keys", [])
            ]

            indexes = [
                IndexDef(
                    name=idx["name"],
                    column_names=idx.get("column_names", []),
                    unique=idx.get("unique", False),
                )
                for idx in t.get("indexes", [])
            ]

            tables.append(TableDef(
                name=t["name"],
                columns=columns,
                foreign_keys=foreign_keys,
                indexes=indexes,
                sample_data=[],
            ))

        return SchemaDef(tables=tables)

    def generate_mermaid_erd(self, raw: dict) -> str:
        """
        Sinh Mermaid ERD string từ raw schema JSON để frontend render diagram.
        """
        lines = ["erDiagram"]

        for table in raw.get("tables", []):
            table_name = table["name"].upper()
            lines.append(f"  {table_name} {{")

            for col in table.get("columns", []):
                constraints = []
                if col.get("is_pk"):
                    constraints.append("PK")
                fk_cols = [fk["column"] for fk in table.get("foreign_keys", [])]
                if col["name"] in fk_cols:
                    constraints.append("FK")
                marker = " ".join(constraints)
                lines.append(f'    {col["type"]} {col["name"]} {marker}')

            lines.append("  }")

        type_map = {
            "many_to_one":  "}o--||",
            "one_to_many":  "||--o{",
            "one_to_one":   "||--||",
            "many_to_many": "}o--o{",
        }

        for rel in raw.get("relationships", []):
            symbol = type_map.get(rel.get("type", ""), "||--o{")
            from_t = rel["from_table"].upper()
            to_t = rel["to_table"].upper()
            desc = rel.get("description", "")
            lines.append(f'  {from_t} {symbol} {to_t} : "{desc}"')

        return "\n".join(lines)


schema_generator_service = SchemaGeneratorService()
~~~

~~~python
import logging
from typing import List, Optional

from app.api.v1.endpoints.auth import get_current_user
from app.models.models import User
from app.services.schema_generator_service import schema_generator_service
from fastapi import APIRouter, Depends, HTTPException, status
from pydantic import BaseModel, Field

router = APIRouter()
logger = logging.getLogger(__name__)


# ── Request / Response schemas ────────────────────────────────────────────────

class ClarificationAnswer(BaseModel):
    question: str
    answer: str


class GenerateSchemaRequest(BaseModel):
    description: str = Field(
        ...,
        min_length=5,
        max_length=2000,
        description="Natural language description of the system to build"
    )
    clarifications: Optional[List[ClarificationAnswer]] = Field(
        default=None,
        description="Optional answers to clarification questions"
    )

    class Config:
        json_schema_extra = {
            "example": {
                "description": "Hệ thống quản lý thư viện cho phép thành viên mượn và trả sách, theo dõi lịch sử và tính phí phạt quá hạn",
                "clarifications": [
                    {
                        "question": "Một bản sao vật lý của sách có được theo dõi riêng không?",
                        "answer": "Có, mỗi quyển sách có nhiều bản copy"
                    }
                ]
            }
        }


class CheckClarificationRequest(BaseModel):
    description: str = Field(..., min_length=5, max_length=2000)


class ClarificationQuestion(BaseModel):
    question: str
    why: str
    options: List[str]


class CheckClarificationResponse(BaseModel):
    needs_clarification: bool
    questions: List[ClarificationQuestion]


class GenerateSchemaResponse(BaseModel):
    system_name: str
    schema_def: dict                # SchemaDef.to_json_dict() — compatible với PUT /connections/{id}/schema
    mermaid_erd: str
    relationships: list
    design_notes: list
    table_count: int
    table_summaries: List[dict]     # [{ name, purpose, column_count, index_count }]


# ── Endpoints ─────────────────────────────────────────────────────────────────

@router.post(
    "/check-clarification",
    response_model=CheckClarificationResponse,
    status_code=status.HTTP_200_OK,
    summary="Check if description needs clarification before generating schema",
)
async def check_clarification(
    request: CheckClarificationRequest,
    current_user: User = Depends(get_current_user),
):
    try:
        result = await schema_generator_service.check_clarification(
            request.description
        )
        return CheckClarificationResponse(
            needs_clarification=result.get("needs_clarification", False),
            questions=[
                ClarificationQuestion(**q)
                for q in result.get("questions", [])
            ],
        )
    except Exception as e:
        logger.error(f"[SCHEMA-GEN] Clarification check failed: {e}")
        # Non-critical: nếu lỗi thì skip clarification
        return CheckClarificationResponse(
            needs_clarification=False,
            questions=[]
        )


@router.post(
    "/generate",
    response_model=GenerateSchemaResponse,
    status_code=status.HTTP_200_OK,
    summary="Generate database schema from natural language description",
)
async def generate_schema(
    request: GenerateSchemaRequest,
    current_user: User = Depends(get_current_user),
):
    try:
        clarifications = (
            [c.model_dump() for c in request.clarifications]
            if request.clarifications else None
        )

        result = await schema_generator_service.generate(
            description=request.description,
            clarifications=clarifications,
        )

        schema_def = result["schema_def"]
        raw = result["raw"]

        mermaid_erd = schema_generator_service.generate_mermaid_erd(raw)

        table_summaries = [
            {
                "name": t["name"],
                "purpose": t.get("purpose", ""),
                "design_rationale": t.get("design_rationale", ""),
                "column_count": len(t.get("columns", [])),
                "index_count": len(t.get("indexes", [])),
            }
            for t in raw.get("tables", [])
        ]

        return GenerateSchemaResponse(
            system_name=result["system_name"],
            schema_def=schema_def.to_json_dict(),
            mermaid_erd=mermaid_erd,
            relationships=result["relationships"],
            design_notes=result["design_notes"],
            table_count=len(schema_def.tables),
            table_summaries=table_summaries,
        )

    except ValueError as e:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail=str(e)
        )
    except Exception as e:
        logger.error(f"[SCHEMA-GEN] Generate failed: {e}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Schema generation failed: {str(e)}"
        )
~~~

### 7. Tạo file mới `frontend/src/components/simulation/GenerateSchemaModal.tsx`

Modal này có 3 bước (step) nội bộ:
- Step 1 `"input"` — user nhập mô tả
- Step 2 `"clarify"` — hiện câu hỏi nếu cần
- Step 3 `"preview"` — preview schema trước khi apply

~~~tsx
import { useState } from 'react';
import {
  X, Sparkles, ChevronRight, ChevronLeft,
  Loader2, Database, AlertCircle, Check
} from 'lucide-react';
import { workspaceService } from '../../services/workspaceService';
import { toast } from 'react-toastify';

type Step = 'input' | 'clarify' | 'preview';

interface ClarificationQuestion {
  question: string;
  why: string;
  options: string[];
}

interface TableSummary {
  name: string;
  purpose: string;
  design_rationale: string;
  column_count: number;
  index_count: number;
}

interface GenerateResult {
  system_name: string;
  schema_def: any;
  mermaid_erd: string;
  relationships: any[];
  design_notes: string[];
  table_count: number;
  table_summaries: TableSummary[];
}

interface Props {
  isOpen: boolean;
  onClose: () => void;
  onApply: (schemaDef: any) => void;  // callback nhận schema_def để apply vào TableEditor state
}

export function GenerateSchemaModal({ isOpen, onClose, onApply }: Props) {
  const [step, setStep] = useState<Step>('input');
  const [description, setDescription] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [questions, setQuestions] = useState<ClarificationQuestion[]>([]);
  const [answers, setAnswers] = useState<Record<number, string>>({});
  const [result, setResult] = useState<GenerateResult | null>(null);

  if (!isOpen) return null;

  const handleCheckAndProceed = async () => {
    if (!description.trim() || description.trim().length < 10) {
      toast.error('Please describe your system in at least 10 characters');
      return;
    }

    setIsLoading(true);
    try {
      const check = await workspaceService.checkSchemaClarification(description);

      if (check.needs_clarification && check.questions.length > 0) {
        setQuestions(check.questions);
        setStep('clarify');
      } else {
        await handleGenerate();
      }
    } catch (err) {
      // Nếu check lỗi thì skip thẳng sang generate
      await handleGenerate();
    } finally {
      setIsLoading(false);
    }
  };

  const handleGenerate = async (withClarifications = false) => {
    setIsLoading(true);
    try {
      const clarifications = withClarifications
        ? questions.map((q, idx) => ({
            question: q.question,
            answer: answers[idx] || '',
          })).filter(c => c.answer)
        : undefined;

      const generated = await workspaceService.generateSchemaFromPrompt({
        description,
        clarifications,
      });

      setResult(generated);
      setStep('preview');
    } catch (err: any) {
      toast.error(err?.response?.data?.detail || 'Schema generation failed');
    } finally {
      setIsLoading(false);
    }
  };

  const handleApply = () => {
    if (!result) return;
    onApply(result.schema_def);
    toast.success(`Schema "${result.system_name}" applied — ${result.table_count} tables created`);
    onClose();
    resetState();
  };

  const resetState = () => {
    setStep('input');
    setDescription('');
    setQuestions([]);
    setAnswers({});
    setResult(null);
  };

  const handleClose = () => {
    onClose();
    resetState();
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50">
      <div className="bg-white dark:bg-gray-900 rounded-xl shadow-2xl w-full max-w-2xl mx-4 max-h-[90vh] flex flex-col">

        {/* Header */}
        <div className="flex items-center justify-between px-6 py-4 border-b border-gray-200 dark:border-gray-700">
          <div className="flex items-center gap-2">
            <Sparkles className="w-5 h-5 text-purple-500" />
            <h2 className="text-lg font-semibold text-gray-900 dark:text-white">
              Generate Schema from Description
            </h2>
          </div>
          {/* Step indicator */}
          <div className="flex items-center gap-1 text-xs text-gray-500">
            {(['input', 'clarify', 'preview'] as Step[]).map((s, idx) => (
              <span key={s} className="flex items-center gap-1">
                <span className={`w-5 h-5 rounded-full flex items-center justify-center text-xs font-bold
                  ${step === s
                    ? 'bg-purple-500 text-white'
                    : (['input', 'clarify', 'preview'].indexOf(step) > idx)
                      ? 'bg-green-500 text-white'
                      : 'bg-gray-200 dark:bg-gray-700 text-gray-500'
                  }`}>
                  {(['input', 'clarify', 'preview'].indexOf(step) > idx) ? '✓' : idx + 1}
                </span>
                {idx < 2 && <span className="text-gray-300">›</span>}
              </span>
            ))}
          </div>
          <button onClick={handleClose} className="text-gray-400 hover:text-gray-600 dark:hover:text-gray-300">
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Body */}
        <div className="flex-1 overflow-y-auto px-6 py-5">

          {/* STEP 1: Input */}
          {step === 'input' && (
            <div className="space-y-4">
              <p className="text-sm text-gray-500 dark:text-gray-400">
                Describe your system in natural language. The AI will generate
                a normalized schema with tables, columns, indexes, and relationships.
              </p>
              <textarea
                className="w-full h-36 px-3 py-2 text-sm border border-gray-300 dark:border-gray-600
                  rounded-lg bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100
                  focus:outline-none focus:ring-2 focus:ring-purple-500 resize-none"
                placeholder={`Examples:\n• Hệ thống quản lý thư viện với sách, thành viên, mượn trả và phạt quá hạn\n• E-commerce platform with products, orders, customers, and inventory tracking\n• Hospital management system with patients, doctors, appointments, and prescriptions`}
                value={description}
                onChange={e => setDescription(e.target.value)}
                autoFocus
              />
              <p className="text-xs text-gray-400">
                {description.length} / 2000 characters
              </p>
            </div>
          )}

          {/* STEP 2: Clarification */}
          {step === 'clarify' && (
            <div className="space-y-5">
              <p className="text-sm text-gray-500 dark:text-gray-400">
                A few quick questions to generate a better schema:
              </p>
              {questions.map((q, idx) => (
                <div key={idx} className="space-y-2">
                  <p className="text-sm font-medium text-gray-800 dark:text-gray-200">
                    {idx + 1}. {q.question}
                  </p>
                  <p className="text-xs text-gray-400 italic">{q.why}</p>
                  {/* Quick option buttons */}
                  <div className="flex flex-wrap gap-2">
                    {q.options.map((opt, oidx) => (
                      <button
                        key={oidx}
                        onClick={() => setAnswers(prev => ({ ...prev, [idx]: opt }))}
                        className={`px-3 py-1 rounded-full text-xs border transition-colors
                          ${answers[idx] === opt
                            ? 'bg-purple-100 border-purple-400 text-purple-700 dark:bg-purple-900 dark:text-purple-300'
                            : 'border-gray-300 dark:border-gray-600 text-gray-600 dark:text-gray-400 hover:border-purple-400'
                          }`}
                      >
                        {opt}
                      </button>
                    ))}
                  </div>
                  {/* Free text override */}
                  <input
                    type="text"
                    placeholder="Or type your own answer..."
                    value={answers[idx] || ''}
                    onChange={e => setAnswers(prev => ({ ...prev, [idx]: e.target.value }))}
                    className="w-full px-3 py-1.5 text-sm border border-gray-300 dark:border-gray-600
                      rounded-lg bg-white dark:bg-gray-800 text-gray-900 dark:text-gray-100
                      focus:outline-none focus:ring-2 focus:ring-purple-500"
                  />
                </div>
              ))}
            </div>
          )}

          {/* STEP 3: Preview */}
          {step === 'preview' && result && (
            <div className="space-y-4">
              <div className="flex items-center gap-2">
                <Check className="w-4 h-4 text-green-500" />
                <p className="text-sm font-semibold text-gray-800 dark:text-gray-200">
                  {result.system_name} — {result.table_count} tables generated
                </p>
              </div>

              {/* Table summaries */}
              <div className="space-y-2">
                {result.table_summaries.map((t, idx) => (
                  <div
                    key={idx}
                    className="flex items-start gap-3 p-3 rounded-lg bg-gray-50 dark:bg-gray-800 border border-gray-200 dark:border-gray-700"
                  >
                    <Database className="w-4 h-4 text-blue-500 mt-0.5 flex-shrink-0" />
                    <div className="flex-1 min-w-0">
                      <div className="flex items-center gap-2 flex-wrap">
                        <span className="text-sm font-mono font-semibold text-blue-600 dark:text-blue-400">
                          {t.name}
                        </span>
                        <span className="text-xs text-gray-400">
                          {t.column_count} cols · {t.index_count} indexes
                        </span>
                      </div>
                      <p className="text-xs text-gray-500 dark:text-gray-400 mt-0.5">{t.purpose}</p>
                      {t.design_rationale && (
                        <p className="text-xs text-gray-400 italic mt-0.5">{t.design_rationale}</p>
                      )}
                    </div>
                  </div>
                ))}
              </div>

              {/* Design notes */}
              {result.design_notes.length > 0 && (
                <div className="p-3 rounded-lg bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800">
                  <p className="text-xs font-semibold text-amber-700 dark:text-amber-400 mb-1">
                    Design Notes
                  </p>
                  <ul className="space-y-1">
                    {result.design_notes.map((note, idx) => (
                      <li key={idx} className="text-xs text-amber-600 dark:text-amber-400">
                        • {note}
                      </li>
                    ))}
                  </ul>
                </div>
              )}

              {/* Warning: sẽ replace schema hiện tại */}
              <div className="flex items-start gap-2 p-3 rounded-lg bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-700">
                <AlertCircle className="w-4 h-4 text-red-500 flex-shrink-0 mt-0.5" />
                <p className="text-xs text-red-600 dark:text-red-400">
                  Applying this schema will <strong>replace your current sandbox schema</strong>.
                  Any existing tables and data will be lost.
                </p>
              </div>
            </div>
          )}

          {/* Loading overlay */}
          {isLoading && (
            <div className="flex flex-col items-center justify-center py-12 gap-3">
              <Loader2 className="w-8 h-8 text-purple-500 animate-spin" />
              <p className="text-sm text-gray-500">
                {step === 'input' ? 'Analyzing description...' : 'Generating schema...'}
              </p>
            </div>
          )}
        </div>

        {/* Footer */}
        <div className="flex items-center justify-between px-6 py-4 border-t border-gray-200 dark:border-gray-700">
          <button
            onClick={step === 'input' ? handleClose : () => setStep(step === 'preview' ? 'clarify' : 'input')}
            disabled={isLoading}
            className="flex items-center gap-1 px-4 py-2 text-sm text-gray-600 dark:text-gray-400
              hover:text-gray-800 dark:hover:text-gray-200 disabled:opacity-50 transition-colors"
          >
            <ChevronLeft className="w-4 h-4" />
            {step === 'input' ? 'Cancel' : 'Back'}
          </button>

          <div className="flex items-center gap-2">
            {step === 'input' && (
              <button
                onClick={handleCheckAndProceed}
                disabled={isLoading || description.trim().length < 10}
                className="flex items-center gap-2 px-5 py-2 bg-purple-600 hover:bg-purple-700
                  disabled:opacity-50 text-white text-sm font-medium rounded-lg transition-colors"
              >
                {isLoading ? <Loader2 className="w-4 h-4 animate-spin" /> : <Sparkles className="w-4 h-4" />}
                Generate Schema
              </button>
            )}

            {step === 'clarify' && (
              <>
                <button
                  onClick={() => handleGenerate(false)}
                  disabled={isLoading}
                  className="px-4 py-2 text-sm text-gray-500 hover:text-gray-700 dark:hover:text-gray-300
                    disabled:opacity-50 transition-colors"
                >
                  Skip questions
                </button>
                <button
                  onClick={() => handleGenerate(true)}
                  disabled={isLoading}
                  className="flex items-center gap-2 px-5 py-2 bg-purple-600 hover:bg-purple-700
                    disabled:opacity-50 text-white text-sm font-medium rounded-lg transition-colors"
                >
                  {isLoading ? <Loader2 className="w-4 h-4 animate-spin" /> : <ChevronRight className="w-4 h-4" />}
                  Generate with answers
                </button>
              </>
            )}

            {step === 'preview' && (
              <button
                onClick={handleApply}
                disabled={isLoading}
                className="flex items-center gap-2 px-5 py-2 bg-green-600 hover:bg-green-700
                  disabled:opacity-50 text-white text-sm font-medium rounded-lg transition-colors"
              >
                <Check className="w-4 h-4" />
                Apply to Sandbox
              </button>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
~~~

#### 8c. Thêm handler `handleApplyGeneratedSchema`

Đặt cùng khu vực với các handler khác trong component:

~~~tsx
const handleApplyGeneratedSchema = (schemaDef: any) => {
  // Convert schema_def (backend format) → SimulationSchema (frontend format với IDs)
  // Tái sử dụng logic parse đã có trong useEffect khi load workspace

  const backendTables = (schemaDef.tables || []) as BackendTable[];
  const tableIdMap = new Map<string, string>();
  const columnIdMap = new Map<string, Map<string, string>>();

  const tablesWithIds = backendTables.map(table => {
    const tableId = uuidv4();
    tableIdMap.set(table.name, tableId);

    const colMap = new Map<string, string>();
    const columns = table.columns.map(col => {
      const colId = uuidv4();
      colMap.set(col.name, colId);
      return {
        id: colId,
        name: col.name,
        type: col.type,
        is_pk: col.is_pk,
        is_nullable: col.is_nullable,
        default: col.default,
      };
    });
    columnIdMap.set(table.name, colMap);

    const indexes = (table.indexes || []).map(idx => ({
      id: uuidv4(),
      name: idx.name,
      columns: (idx.column_names || [])
        .map(colName => colMap.get(colName))
        .filter((id): id is string => !!id),
      unique: idx.unique || false,
    }));

    return { id: tableId, name: table.name, columns, indexes, foreign_keys: table.foreign_keys || [], sample_data: [] };
  });

  // Restore FK targets
  const newSchema: SimulationSchema = {
    is_simulation: true,
    tables: tablesWithIds.map(table => ({
      ...table,
      columns: table.columns.map(col => {
        const fkDef = table.foreign_keys.find(fk => fk.column === col.name);
        if (fkDef) {
          const refTableId = tableIdMap.get(fkDef.ref_table);
          const refColId = columnIdMap.get(fkDef.ref_table)?.get(fkDef.ref_column);
          if (refTableId && refColId) {
            return { ...col, fk_target: { table_id: refTableId, column_id: refColId } };
          }
        }
        return col;
      }),
    })),
  };

  setSchema(newSchema);
  setHasUnsavedChanges(true);

  // Auto-select first table
  if (newSchema.tables.length > 0) {
    setSelectedTableId(newSchema.tables[0].id);
  }
};
~~~
